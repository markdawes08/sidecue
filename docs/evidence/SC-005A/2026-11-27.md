    '--no-build', '--no-restore', '--logger', 'trx;LogFileName=sc-005a-desktop.trx',
    '--results-directory', (Join-Path $repoRoot 'artifacts\test-results'))

$env:SIDECUE_RUN_SCREENSHOT_NATIVE_PROBE = '1'
try {
    Invoke-SidecueDotnet @('test', 'tests/Sidecue.Desktop.Tests', '-c', 'Release',
        '--no-build', '--no-restore', '--filter', 'FullyQualifiedName~ScreenshotNativeProbeTests',
        '--logger', 'trx;LogFileName=sc-005a-native-standard-retry.trx',
        '--results-directory', (Join-Path $repoRoot 'artifacts\test-results'))
} finally { Remove-Item Env:SIDECUE_RUN_SCREENSHOT_NATIVE_PROBE }
