Actual app smoke command:

```powershell
$env:DOTNET_ROOT = Join-Path (Get-Location) 'artifacts\dotnet'
$env:DOTNET_ROOT_X64 = $env:DOTNET_ROOT
$appPath = Join-Path (Get-Location) 'src\Sidecue.Desktop\bin\Release\net10.0-windows10.0.26100.0\Sidecue.Desktop.exe'
$shellProcess = Start-Process -FilePath $appPath -WindowStyle Hidden -PassThru
try {
    if (-not $shellProcess.WaitForInputIdle(10000)) { throw 'Shell did not become input-idle' }
    $shellProcess.Refresh()
    if ($shellProcess.HasExited -or $shellProcess.MainWindowHandle -eq [IntPtr]::Zero -or $shellProcess.MainWindowTitle -ne 'Sidecue') {
        throw 'Shell did not create its expected window'
    }
    if (-not $shellProcess.CloseMainWindow()) { throw 'Shell close request failed' }
    if (-not $shellProcess.WaitForExit(5000) -or $shellProcess.ExitCode -ne 0) {
        throw 'Shell did not shut down normally'
    }
} finally {
    if (-not $shellProcess.HasExited) { Stop-Process -Id $shellProcess.Id }
    $shellProcess.Dispose()
}
```
